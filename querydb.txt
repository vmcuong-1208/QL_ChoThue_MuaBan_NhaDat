-- Tạo cơ sở dữ liệu
CREATE DATABASE IF NOT EXISTS QUANLYTHUEMUABANNHADAT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

-- Sử dụng cơ sở dữ liệu vừa tạo
USE QUANLYTHUEMUABANNHADAT;

-- Bảng Người Dùng
CREATE TABLE NGUOIDUNG (
    MaNgDung INT AUTO_INCREMENT PRIMARY KEY,
    TenDangNhap VARCHAR(50),
    Email VARCHAR(100) NOT NULL UNIQUE,
    MatKhau VARCHAR(255) NULL,
    SDT VARCHAR(15),
    DiaChi TEXT,
    LoaiTaiKhoan ENUM('thường', 'môi giới', 'chủ đầu tư', 'admin') DEFAULT 'thường',
    HoTen VARCHAR(100),
    AnhDaiDien VARCHAR(255),
    MoTaBanThan TEXT,
    TrangThai ENUM('hoạt động', 'khóa', 'chờ xác thực') DEFAULT 'chờ xác thực',
    GoogleId VARCHAR(255) NULL UNIQUE,
    NgayTao DATETIME DEFAULT CURRENT_TIMESTAMP,
    NgayCapNhat DATETIME ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT chk_auth_method 
    CHECK (MatKhau IS NOT NULL OR GoogleId IS NOT NULL) -- Constraint đảm bảo ít nhất một trong hai có giá trị
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Bảng Loại nhà đất
CREATE TABLE LOAINHADAT (
    MaLoaiND INT AUTO_INCREMENT PRIMARY KEY,
    TenLoaiND VARCHAR(100) NOT NULL UNIQUE,
    MoTa TEXT
);
-- Bảng Thông tin nhà đất 
CREATE TABLE THONGTINNHADAT (
    MaTTND INT AUTO_INCREMENT PRIMARY KEY,
    MaNgDung INT NOT NULL,
    TieuDe VARCHAR(255) NOT NULL,
    MoTa TEXT,
    MaLoaiND INT NOT NULL,
    Gia DECIMAL(15,2) NOT NULL,
    DienTich DECIMAL(10,2) NOT NULL,
    DiaChi VARCHAR(255) NOT NULL,
    MaTinh  VARCHAR(20) NOT NULL,
    MaHuyen  VARCHAR(20) NOT NULL,
    MaXa  VARCHAR(20) NOT NULL,
    SoLuotXem INT UNSIGNED DEFAULT 0,
    TrangThai ENUM('đang bán', 'cho thuê', 'đã bán', 'đã cho thuê','chờ xác thực','đã xác thực') DEFAULT 'đang bán',
    NgayTao DATETIME DEFAULT CURRENT_TIMESTAMP,
    NgayCapNhat DATETIME ON UPDATE CURRENT_TIMESTAMP,
    GoogleMapIframe TEXT,
    TrangThaiXacThuc ENUM('chưa xác thực', 'chờ xác thực', 'đã xác thực', 'từ chối') DEFAULT 'chưa xác thực';
    FOREIGN KEY (MaNgDung) REFERENCES NGUOIDUNG(MaNgDung) ON DELETE CASCADE,
	FOREIGN KEY (MaLoaiND) REFERENCES LOAINHADAT(MaLoaiND) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Tạo bảng thông tin xác thực
CREATE TABLE THONGTINXACTHUC (
    MaXacThuc INT AUTO_INCREMENT PRIMARY KEY,
    MaTTND INT NOT NULL,
    -- Các trường admin sẽ check
    CoSoDoSoHong BOOLEAN DEFAULT FALSE,
    DaCungCapBanChup BOOLEAN DEFAULT FALSE,
    DungDiaChi BOOLEAN DEFAULT FALSE,
    BDSTonTaiDungDiaChi BOOLEAN DEFAULT FALSE,
    DungHinhAnhDacDiem BOOLEAN DEFAULT FALSE,
    DaDoiChieuVoiSoDo BOOLEAN DEFAULT FALSE,
    CoGiayXacNhanKhongTraanhChap BOOLEAN DEFAULT FALSE,
    DaKiemTraQuyHoach BOOLEAN DEFAULT FALSE,
    -- Trường user tự check
    CamKetCapNhatDinhKy BOOLEAN DEFAULT FALSE,
    -- Files upload
    FileSoDo VARCHAR(255),
    VideoXacThuc VARCHAR(255),
    FileGiayXacNhan VARCHAR(255),
    FileQuyHoach VARCHAR(255),
    GhiChuNguoiBan TEXT,
    NgayGui DATETIME DEFAULT CURRENT_TIMESTAMP,
    NgayXacThuc DATETIME NULL,
    NguoiXacThuc INT NULL,
    GhiChuAdmin TEXT,
    FOREIGN KEY (MaTTND) REFERENCES THONGTINNHADAT(MaTTND) ON DELETE CASCADE,
    FOREIGN KEY (NguoiXacThuc) REFERENCES NGUOIDUNG(MaNgDung) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE TINHTHANH (
    MaTinh VARCHAR(20) PRIMARY KEY,
    TenTinh VARCHAR(100) NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE QUANHUYEN (
    MaHuyen VARCHAR(20) PRIMARY KEY,
    TenHuyen VARCHAR(100) NOT NULL,
    MaTinh VARCHAR(20),
    FOREIGN KEY (MaTinh) REFERENCES TINHTHANH(MaTinh)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE PHUONGXA (
    MaXa VARCHAR(20) PRIMARY KEY,
    TenXa VARCHAR(100) NOT NULL,
    MaHuyen VARCHAR(20),
    FOREIGN KEY (MaHuyen) REFERENCES QUANHUYEN(MaHuyen)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Bảng Hình Ảnh BĐS
CREATE TABLE HINHANHBDS (
    MaHA INT AUTO_INCREMENT PRIMARY KEY,
    MaTTND INT NOT NULL,
    DuongDanAnh VARCHAR(255) NOT NULL,
    MoTaAnh VARCHAR(255),
    FOREIGN KEY (MaTTND) REFERENCES THONGTINNHADAT(MaTTND) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Bảng Yêu Thích
CREATE TABLE YEUTHICH (
    MaYT INT AUTO_INCREMENT PRIMARY KEY,
    MaNgDung INT NOT NULL,
    MaTTND INT NOT NULL,
    NgayTao DATETIME DEFAULT CURRENT_TIMESTAMP,
    GhiChu VARCHAR(255),
    FOREIGN KEY (MaNgDung) REFERENCES NGUOIDUNG(MaNgDung) ON DELETE CASCADE,
    FOREIGN KEY (MaTTND) REFERENCES THONGTINNHADAT(MaTTND) ON DELETE CASCADE,
    UNIQUE KEY (MaNgDung, MaTTND)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Bảng Tin Nhắn
CREATE TABLE TINNHAN (
    MaTN INT AUTO_INCREMENT PRIMARY KEY,
    MaNgMua INT NOT NULL,
    MaNgBan INT NOT NULL,
    MaTTND INT,
    NoiDung TEXT NOT NULL,
    TrangThaiDoc BOOLEAN DEFAULT FALSE,
    NgayGui DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (MaNgMua) REFERENCES NGUOIDUNG(MaNgDung) ON DELETE CASCADE,
    FOREIGN KEY (MaNgBan) REFERENCES NGUOIDUNG(MaNgDung) ON DELETE CASCADE,
    FOREIGN KEY (MaTTND) REFERENCES THONGTINNHADAT(MaTTND) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Tạo bảng lịch sử xem nếu chưa có
CREATE TABLE IF NOT EXISTS LICHSUXEM (
    MaLichSu INT AUTO_INCREMENT PRIMARY KEY,
    MaNgDung INT NOT NULL,
    MaTTND INT NOT NULL,
    NgayXem DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (MaNgDung) REFERENCES NGUOIDUNG(MaNgDung) ON DELETE CASCADE,
    FOREIGN KEY (MaTTND) REFERENCES THONGTINNHADAT(MaTTND) ON DELETE CASCADE,
    UNIQUE KEY unique_user_property (MaNgDung, MaTTND)
);
CREATE TABLE IF NOT EXISTS GIAODICH (
    MaGD INT AUTO_INCREMENT PRIMARY KEY,
    MaNgDung INT NOT NULL,
    OrderId VARCHAR(100) UNIQUE NOT NULL,
    TransId VARCHAR(100),
    Amount DECIMAL(15,2) NOT NULL,
    OrderInfo TEXT,
    UpgradeType VARCHAR(50),
    TrangThai ENUM('PENDING', 'SUCCESS', 'FAILED', 'CANCELLED') DEFAULT 'PENDING',
    NgayTao DATETIME DEFAULT CURRENT_TIMESTAMP,
    NgayCapNhat DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (MaNgDung) REFERENCES NGUOIDUNG(MaNgDung) ON DELETE CASCADE
);
-- Tạo bảng OTP cho chức năng quên mật khẩu
CREATE TABLE IF NOT EXISTS OTP (
    MaOTP INT AUTO_INCREMENT PRIMARY KEY,
    Email VARCHAR(100) NOT NULL,
    OTPCode VARCHAR(6) NOT NULL,
    ThoiGianTao DATETIME DEFAULT CURRENT_TIMESTAMP,
    ThoiGianHetHan DATETIME NOT NULL,
    DaSuDung BOOLEAN DEFAULT FALSE,
    INDEX idx_email (Email),
    INDEX idx_otp_code (OTPCode),
    INDEX idx_expiry (ThoiGianHetHan),
    FOREIGN KEY (Email) REFERENCES NGUOIDUNG(Email) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Xóa OTP cũ đã hết hạn (chạy định kỳ)
CREATE EVENT IF NOT EXISTS delete_expired_otp
ON SCHEDULE EVERY 1 DAY
STARTS CURRENT_TIMESTAMP
DO
    DELETE FROM OTP WHERE ThoiGianHetHan < NOW();
-- Tạo index để tối ưu hóa truy vấn
CREATE INDEX idx_property_price ON THONGTINNHADAT(Gia);
CREATE INDEX idx_property_area ON THONGTINNHADAT(DienTich);
CREATE INDEX idx_property_user ON THONGTINNHADAT(MaNgDung);
CREATE INDEX idx_property_loai ON THONGTINNHADAT(MaLoaiND);
CREATE INDEX idx_yeuthich_user ON YEUTHICH(MaNgDung);
CREATE INDEX idx_yeuthich_property ON YEUTHICH(MaTTND);
CREATE INDEX idx_danhgia_property ON DANHGIA(MaTTND);
CREATE INDEX idx_verification_status ON THONGTINNHADAT(TrangThaiXacThuc);
CREATE INDEX idx_verification_property ON THONGTINXACTHUC(MaTTND);
CREATE INDEX idx_verification_date ON THONGTINXACTHUC(NgayGui);
ADD INDEX idx_google_id (GoogleId);
